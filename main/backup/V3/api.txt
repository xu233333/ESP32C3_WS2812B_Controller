/api/get_config         [GET] -> 获取配置
""
->
{
    "AP_Mode": ? (BOOL),
    "Wifi_SSID": "" (STRING),
    "Wifi_Password": "" (STRING),
    "AP_SSID": "" (STRING),
    "AP_Password": "" (STRING),
    "LED_Pin": ? (INT),
    "LED_Block_Count": ? (INT),
    "DefaultBrightness": ? (INT)
}

/api/set_config         [POST] -> 设置配置
{
    "AP_Mode": ? (BOOL),
    "Wifi_SSID": "" (STRING),
    "Wifi_Password": "" (STRING),
    "AP_SSID": "" (STRING),
    "AP_Password": "" (STRING),
    "LED_Pin": ? (INT),
    "LED_Block_Count": ? (INT),
    "DefaultBrightness": ? (INT)
}
->
"OK"
